/*
# Windows 7 Style WER Dialogs
Restores the Windows 7 WER (Windows Error Reporting) dialog behavior in later versions of Windows 10. Supports 64-bit and 32-bit applications.

### DefaultConsent=1 (Ask before checking)
Shows the wide dialog with:
- "Check for a solution and close the program"
- "Close the program"
- "Wait for the program to respond" (for not responding)
- "View problem details" footer

### DefaultConsent values 2, 3 and 4
Restores the "has stopped working" dialog that later versions of Windows 10 had suppressed.

## Registry
Set `DefaultConsent` (REG_DWORD) to `1`, `2`, `3` or `4` in:
- `HKCU\SOFTWARE\Microsoft\Windows\Windows Error Reporting\Consent` (per-user setting)
- `HKLM\SOFTWARE\Microsoft\Windows\Windows Error Reporting\Consent` (machine setting)

You can also natively adjust this setting without changing registry in `Action Center > Maintenance > Check for solutions to problem reports > Settings`. (restoration required as not present in stock Windows 10)

## Compatibility
Tested on Windows 10 IoT LTSC 21H2. Should work on different Windows versions, although not guaranteed.
*/

// ============================================================================
// Pattern scanner
// ============================================================================
static void* PatternScan(HMODULE hModule, const BYTE* pattern,
                          const char* mask, size_t len) {
    BYTE* base = (BYTE*)hModule;
    IMAGE_DOS_HEADER* dos = (IMAGE_DOS_HEADER*)base;
    IMAGE_NT_HEADERS* nt = (IMAGE_NT_HEADERS*)(base + dos->e_lfanew);
    IMAGE_SECTION_HEADER* sec = IMAGE_FIRST_SECTION(nt);
    for (WORD i = 0; i < nt->FileHeader.NumberOfSections; i++, sec++) {
        if (!(sec->Characteristics & IMAGE_SCN_MEM_EXECUTE)) continue;
        BYTE* start = base + sec->VirtualAddress;
        DWORD size = sec->Misc.VirtualSize;
        for (DWORD j = 0; j + len <= size; j++) {
            bool ok = true;
            for (size_t k = 0; k < len && ok; k++)
                if (mask[k] == 'x' && start[j + k] != pattern[k]) ok = false;
            if (ok) return start + j;
        }
    }
    return nullptr;
}

static void* Resolve(HMODULE hMod, const BYTE* pat, const char* mask,
                      size_t len, DWORD rva) {
    void* addr = PatternScan(hMod, pat, mask, len);
    if (!addr && rva) addr = (BYTE*)hMod + rva;
    return addr;
}

    long r = WerpGetReportFlags_orig(p1, p2);

// ============================================================================
// Hook: WerpGetUIParamByIndex
// ============================================================================
typedef long (WINAPI *WerpGetUIParamByIndex_t)(void*, int, unsigned short**);
WerpGetUIParamByIndex_t WerpGetUIParamByIndex_orig;

long WINAPI WerpGetUIParamByIndex_hook(void* p1, int idx, unsigned short** pp) {
    long r = WerpGetUIParamByIndex_orig(p1, idx, pp);
    if (g_defaultConsent != 1 || !pp) return r;

    BOOL online = IsOnline();
    wchar_t* text = nullptr;

    if (idx == 5 || idx == 6) {
        if (g_origReportType == 2) {
            unsigned long fl = 0;
            WerpGetReportFlags_hook(p1, &fl);
            text = online ? g_crashBtnOn[fl & 3] : g_crashBtnOff[fl & 3];
        } else if (idx == 5 || idx == 6) {
            if (g_targetIsExplorer && g_hangRestartBtnOn[0])
                text = online ? g_hangRestartBtnOn : g_hangRestartBtnOff;
            else
                text = online ? g_hangBtnOn : g_hangBtnOff;
        }
    } else if (idx == 4) {
        if (g_origReportType == 2) {
            unsigned long fl = 0;
            WerpGetReportFlags_hook(p1, &fl);
            text = online ? g_crashCtOn[fl & 3] : g_crashCtOff[fl & 3];
        } else {
            if (g_targetIsExplorer && g_hangRestartCtOn[0])
                text = online ? g_hangRestartCtOn : g_hangRestartCtOff;
            else
                text = online ? g_hangCtOn : g_hangCtOff;
        }
    }

    if (text && text[0]) { *pp = (unsigned short*)text; return 0; }
    return r;
}

    int* pType = (int*)((char*)pThis + OFF_REPORT_TYPE);
    int* pState = (int*)((char*)pThis + OFF_UI_STATE);
    int origType = *pType;
    g_origReportType = origType;

    if (g_defaultConsent == 1) {
        if (origType == 2 || origType == 4) *pType = 3;
        if (*pState != 0) *pState = 0;
    }

    USER_SELECTION result = GetInitDlg_orig(pThis);
    *pType = origType;
    return result;
}

// Hook: ShouldUseHeadLessUIMode
// ============================================================================
typedef int (__thiscall *Headless_t)(void*);
Headless_t Headless_orig;

int __thiscall Headless_hook(void* pThis) {
    int r = Headless_orig(pThis);
    if (r != 0) {
        LPWSTR cmd = GetCommandLineW();
        LPCWSTR pf = cmd ? wcsstr(cmd, L"-p ") : NULL;
        if (!pf) pf = cmd ? wcsstr(cmd, L"/p ") : NULL;
        if (pf) {
            DWORD pid = (DWORD)_wtoi(pf + 3);
            wchar_t path[MAX_PATH], file[MAX_PATH];
            GetTempPathW(MAX_PATH, path);
            wsprintfW(file, L"%sWerConsentUI_%d.tmp", path, pid);
            HANDLE hf = CreateFileW(file, GENERIC_WRITE, 0, NULL,
                                    CREATE_NEW, FILE_ATTRIBUTE_TEMPORARY, NULL);
            if (hf != INVALID_HANDLE_VALUE) { CloseHandle(hf); r = 0; }
            else DeleteFileW(file);
        }
    }

    // When forcing UI, ensure werui.dll is loaded and hooked
    if (r == 0 && !g_WeruiDllHooked) {
        HMODULE h = GetModuleHandleW(L"werui.dll");
        if (!h) h = LoadLibraryW(L"werui.dll");
        if (h) { InstallWeruiDllHooks(h); Wh_ApplyHookOperations(); }
    }

    if (r == 0) {
#ifdef _WIN64
        int* pSettings = *(int**)((char*)pThis + OFF_SETTINGS_PTR);
        if (pSettings) *(int*)((char*)pSettings + 0x750) = 0;
#else
        int* pSettings = *(int**)((char*)pThis + OFF_SETTINGS_PTR);
        if (pSettings) *(int*)((char*)pSettings + 0x420) = 0;
#endif
    }

    return r;
}

// ============================================================================
// Crash/hang dialog repositioning
// ============================================================================
struct FindWndData { DWORD pid; HWND hwnd; };

static BOOL CALLBACK FindMainWnd(HWND hwnd, LPARAM lp) {
    FindWndData* d = (FindWndData*)lp;
    DWORD p = 0;
    GetWindowThreadProcessId(hwnd, &p);
    if (p == d->pid && IsWindowVisible(hwnd) && !GetWindow(hwnd, GW_OWNER))
        { d->hwnd = hwnd; return FALSE; }
    return TRUE;
}

static void RepositionOverTarget(HWND hDlg) {
    if (!hDlg || !IsWindow(hDlg)) return;

    LPWSTR cmd = GetCommandLineW();
    if (!cmd) return;
    if (!wcsstr(cmd, L"-u ") && !wcsstr(cmd, L"/u ") &&
        !wcsstr(cmd, L"-h ") && !wcsstr(cmd, L"/h ")) return;

    LPCWSTR pf = wcsstr(cmd, L"-p ");
    if (!pf) pf = wcsstr(cmd, L"/p ");
    if (!pf) return;

    DWORD pid = (DWORD)_wtoi(pf + 3);
    if (!pid) return;

    FindWndData d = { pid, NULL };
    EnumWindows(FindMainWnd, (LPARAM)&d);
    if (!d.hwnd) return;

    RECT tr, dr;
    if (!GetWindowRect(d.hwnd, &tr) || !GetWindowRect(hDlg, &dr)) return;

    int dw = dr.right - dr.left, dh = dr.bottom - dr.top;
    int tbH = GetSystemMetrics(SM_CYCAPTION) + GetSystemMetrics(SM_CYFRAME);
    int fW = GetSystemMetrics(SM_CXFRAME);
    int cL = tr.left + fW, cT = tr.top + tbH;
    int cW = (tr.right - fW) - cL;
    int cH = (tr.bottom - GetSystemMetrics(SM_CYFRAME)) - cT;

    int x, y;
    if (cH < dh + 20 || cW < dw + 20) {
        x = tr.left + ((tr.right - tr.left) - dw) / 2;
        y = cT + 19;
    } else {
        x = cL + (cW - dw) / 2;
        y = cT + (cH - dh) / 2;
    }

    HMONITOR hm = MonitorFromRect(&tr, MONITOR_DEFAULTTONEAREST);
    MONITORINFO mi = { sizeof(mi) };
    if (GetMonitorInfoW(hm, &mi)) {
        RECT w = mi.rcWork;
        if (x < w.left) x = w.left;
        if (y < w.top) y = w.top;
        if (x + dw > w.right) x = w.right - dw;
        if (y + dh > w.bottom) y = w.bottom - dh;
    }

    MoveWindow(hDlg, x, y, dw, dh, TRUE);
    SetWindowPos(hDlg, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE);
}

// UpdateUIForStateRestarting function pointer

typedef long (__thiscall *UpdateUIForStateRestarting_t)(void*);
UpdateUIForStateRestarting_t UpdateUIForStateRestarting_func = nullptr;

static DWORD WINAPI RestartAutoCloseThread(LPVOID lpParam) {
    Sleep(2000);
    HANDLE* params = (HANDLE*)lpParam;
    HANDLE hEvent = params[0];
    delete[] params;
    if (hEvent) SetEvent(hEvent);
    Sleep(500);
    ExitProcess(0);
    return 0;
}

// ============================================================================
// Hook: UpdateUIForStateClosing
// ============================================================================
typedef long (__thiscall *UpdateUIForStateClosing_t)(void*);
UpdateUIForStateClosing_t UpdateUIForStateClosing_orig = nullptr;

static BOOL g_progDialogShown = FALSE;

long __thiscall UpdateUIForStateClosing_hook(void* pThis) {
    if (g_targetIsExplorer && UpdateUIForStateRestarting_func && g_progDialogShown) {
        UpdateUIForStateRestarting_func(pThis);
        HANDLE* params = new HANDLE[1];
#ifdef _WIN64
        params[0] = *(HANDLE*)((char*)pThis + 0xC0);
#else
        params[0] = *(HANDLE*)((char*)pThis + 0x88);
#endif
        HANDLE hThread = CreateThread(NULL, 0, RestartAutoCloseThread, params, 0, NULL);
        if (hThread) CloseHandle(hThread);
        return 0;
    }
    return UpdateUIForStateClosing_orig(pThis);
}

typedef long (__thiscall *OnInitProg_t)(void*, HWND, int);
OnInitProg_t OnInitProg_orig;

static LRESULT CALLBACK RestartSubclassProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp,
                                             UINT_PTR uIdSubclass, DWORD_PTR dwRefData) {
    if (msg == 0x465 && g_targetIsExplorer && UpdateUIForStateRestarting_func && dwRefData) {
        void* pThis = (void*)dwRefData;
        RemoveWindowSubclass(hwnd, RestartSubclassProc, uIdSubclass);
        UpdateUIForStateRestarting_func(pThis);
        HANDLE* params = new HANDLE[1];
#ifdef _WIN64
        params[0] = *(HANDLE*)((char*)pThis + 0xC0);
#else
        params[0] = *(HANDLE*)((char*)pThis + 0x88);
#endif
        HANDLE hThread = CreateThread(NULL, 0, RestartAutoCloseThread, params, 0, NULL);
        if (hThread) CloseHandle(hThread);
        return 0;
    }
    if (msg == 0x401 && g_targetIsExplorer && UpdateUIForStateClosing_orig && dwRefData) {
        void* pThis = (void*)dwRefData;
        RemoveWindowSubclass(hwnd, RestartSubclassProc, uIdSubclass);
        g_progDialogShown = TRUE;
        UpdateUIForStateClosing_hook(pThis);
        return 0;
    }
    return DefSubclassProc(hwnd, msg, wp, lp);
}

long __thiscall OnInitProg_hook(void* pThis, HWND hDlg, int p2) {
    long r = OnInitProg_orig(pThis, hDlg, p2);
    RepositionOverTarget(hDlg);
    g_progDialogShown = TRUE;

    // For explorer: subclass the dialog
    if ((p2 == 0 || p2 == 1) && g_targetIsExplorer && UpdateUIForStateRestarting_func && hDlg) {
        SetWindowSubclass(hDlg, RestartSubclassProc, 0xAE01, (DWORD_PTR)pThis);
    }

    return r;
}

// ============================================================================
// DLL loading hooks (LoadLibrary + LdrLoadDll)
// ============================================================================
typedef HMODULE (WINAPI *LoadLibraryExW_t)(LPCWSTR, HANDLE, DWORD);
LoadLibraryExW_t LoadLibraryExW_orig;
typedef HMODULE (WINAPI *LoadLibraryW_t)(LPCWSTR);
LoadLibraryW_t LoadLibraryW_orig;

typedef struct _MY_UNICODE_STRING {
    USHORT Length; USHORT MaximumLength; PWSTR Buffer;
} MY_UNICODE_STRING;
typedef LONG (__stdcall *LdrLoadDll_t)(PWSTR, ULONG*, MY_UNICODE_STRING*, HMODULE*);
LdrLoadDll_t LdrLoadDll_orig;

static void OnDllLoaded(HMODULE hMod, LPCWSTR name) {
    if (!hMod || !name) return;
    LPCWSTR fn = wcsrchr(name, L'\\');
    fn = fn ? fn + 1 : name;

    if (!g_werDllHooked && _wcsicmp(fn, L"wer.dll") == 0) {
        InstallWerDllHooks(hMod);
        Wh_ApplyHookOperations();
    } else if (!g_WeruiDllHooked && _wcsicmp(fn, L"werui.dll") == 0) {
        InstallWeruiDllHooks(hMod);
        Wh_ApplyHookOperations();
    }
}

HMODULE WINAPI LoadLibraryExW_hook(LPCWSTR n, HANDLE h, DWORD f) {
    HMODULE m = LoadLibraryExW_orig(n, h, f);
    if (!(f & (LOAD_LIBRARY_AS_DATAFILE | LOAD_LIBRARY_AS_IMAGE_RESOURCE)))
        OnDllLoaded(m, n);
    return m;
}
HMODULE WINAPI LoadLibraryW_hook(LPCWSTR n) {
    HMODULE m = LoadLibraryW_orig(n);
    OnDllLoaded(m, n);
    return m;
}

LONG __stdcall LdrLoadDll_hook(PWSTR sp, ULONG* dc, MY_UNICODE_STRING* dn, HMODULE* ba) {
    LONG s = LdrLoadDll_orig(sp, dc, dn, ba);
    if (s == 0 && ba && *ba && dn && dn->Buffer) {
        LPCWSTR fn = wcsrchr(dn->Buffer, L'\\');
        fn = fn ? fn + 1 : dn->Buffer;
        if (!g_WeruiDllHooked && _wcsnicmp(fn, L"werui", 5) == 0) {
            InstallWeruiDllHooks(*ba);
            Wh_ApplyHookOperations();
        } else if (!g_werDllHooked && _wcsnicmp(fn, L"wer.", 4) == 0) {
            InstallWerDllHooks(*ba);
            Wh_ApplyHookOperations();
        }
    }
    return s;
}

static void LoadMuiStrings() {
    HMODULE h = NULL;

#ifdef _WIN64
    // 64-bit
    h = LoadLibraryExW(L"werui.dll", NULL, LOAD_LIBRARY_AS_DATAFILE);
#else
    // 32-bit
    {
        wchar_t locale[16], muiPath[MAX_PATH];
        GetLocaleInfoW(LOCALE_USER_DEFAULT, LOCALE_SNAME, locale, 16);
        wsprintfW(muiPath, L"C:\\Windows\\Sysnative\\%s\\werui.dll.mui", locale);
        h = LoadLibraryExW(muiPath, NULL, LOAD_LIBRARY_AS_DATAFILE);
        if (!h) {
            h = LoadLibraryExW(L"C:\\Windows\\Sysnative\\en-US\\werui.dll.mui",
                               NULL, LOAD_LIBRARY_AS_DATAFILE);
        }
        // Fallback
        if (!h) {
            h = LoadLibraryExW(L"werui.dll", NULL, LOAD_LIBRARY_AS_DATAFILE);
        }
    }
#endif

    if (!h) return;
    // Hang dialog strings
    LoadStringW(h, 0xFDB, g_hangBtnOn, 256);
    LoadStringW(h, 0xFC0, g_hangBtnOff, 256);
    LoadStringW(h, 0xFBD, g_hangCtOn, 512);
    LoadStringW(h, 0xFBE, g_hangCtOff, 512);
    // Crash dialog button text
    LoadStringW(h, 0xEE,  g_crashBtnOn[0], 256);
    LoadStringW(h, 0x101, g_crashBtnOn[1], 256);
    LoadStringW(h, 0x102, g_crashBtnOn[2], 256);
    LoadStringW(h, 0x103, g_crashBtnOn[3], 256);
    LoadStringW(h, 0xEF,  g_crashBtnOff[0], 256);
    LoadStringW(h, 0x104, g_crashBtnOff[1], 256);
    LoadStringW(h, 0x105, g_crashBtnOff[2], 256);
    LoadStringW(h, 0x106, g_crashBtnOff[3], 256);
    // Crash dialog content text
    LoadStringW(h, 0xF9, g_crashCtOn[0], 512);
    LoadStringW(h, 0xFA, g_crashCtOn[1], 512);
    LoadStringW(h, 0xFB, g_crashCtOn[2], 512);
    LoadStringW(h, 0xFC, g_crashCtOn[3], 512);
    LoadStringW(h, 0xFD, g_crashCtOff[0], 512);
    LoadStringW(h, 0xFE, g_crashCtOff[1], 512);

    FreeLibrary(h);
}

BOOL Wh_ModInit() {
    Wh_Log(L"Mod initializing...");

    // Read DefaultConsent: HKCU first, HKLM if not exists
    DWORD cb = sizeof(g_defaultConsent);
    DWORD regFlags = RRF_RT_REG_DWORD | RRF_SUBKEY_WOW6464KEY;
    if (RegGetValueW(HKEY_CURRENT_USER,
            L"SOFTWARE\\Microsoft\\Windows\\Windows Error Reporting\\Consent",
            L"DefaultConsent", regFlags, NULL,
            &g_defaultConsent, &cb) != ERROR_SUCCESS) {
        cb = sizeof(g_defaultConsent);
        RegGetValueW(HKEY_LOCAL_MACHINE,
            L"SOFTWARE\\Microsoft\\Windows\\Windows Error Reporting\\Consent",
            L"DefaultConsent", regFlags, NULL,
            &g_defaultConsent, &cb);
    }

    // If consent not found, try reading from the target process's user registry via token impersonation.
    if (g_defaultConsent < 1 || g_defaultConsent > 4) {
        LPWSTR cmdChk = GetCommandLineW();
        LPCWSTR pfChk = cmdChk ? wcsstr(cmdChk, L"-p ") : NULL;
        if (!pfChk) pfChk = cmdChk ? wcsstr(cmdChk, L"/p ") : NULL;
        if (pfChk) {
            DWORD tgtPid = (DWORD)_wtoi(pfChk + 3);
            if (tgtPid) {
                HANDLE hProc = OpenProcess(PROCESS_QUERY_INFORMATION, FALSE, tgtPid);
                if (hProc) {
                    HANDLE hToken = NULL;
                    if (OpenProcessToken(hProc, TOKEN_READ | TOKEN_IMPERSONATE | TOKEN_DUPLICATE, &hToken)) {
                        if (ImpersonateLoggedOnUser(hToken)) {
                            HKEY hk = NULL;
                            if (RegOpenCurrentUser(KEY_READ, &hk) == ERROR_SUCCESS) {
                                HKEY hConsent = NULL;
                                if (RegOpenKeyExW(hk,
                                        L"SOFTWARE\\Microsoft\\Windows\\Windows Error Reporting\\Consent",
                                        0, KEY_READ, &hConsent) == ERROR_SUCCESS) {
                                    cb = sizeof(g_defaultConsent);
                                    RegQueryValueExW(hConsent, L"DefaultConsent", NULL, NULL,
                                                     (LPBYTE)&g_defaultConsent, &cb);
                                    RegCloseKey(hConsent);
                                }
                                RegCloseKey(hk);
                            }
                            RevertToSelf();
                        }
                        CloseHandle(hToken);
                    }
                    CloseHandle(hProc);
                }
            }
        }
    }

    Wh_Log(L"DefaultConsent=%d", g_defaultConsent);
    if (g_defaultConsent < 1 || g_defaultConsent > 4) return FALSE;

    // Detect if the target process is explorer.exe
    LPWSTR cmd = GetCommandLineW();
    if (cmd) {
        LPCWSTR pf = wcsstr(cmd, L"-p ");
        if (!pf) pf = wcsstr(cmd, L"/p ");
        if (pf) {
            DWORD pid = (DWORD)_wtoi(pf + 3);
            if (pid) {
                HANDLE hProc = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION,
                                           FALSE, pid);
                if (hProc) {
                    wchar_t exePath[MAX_PATH] = {0};
                    DWORD sz = MAX_PATH;
                    if (QueryFullProcessImageNameW(hProc, 0, exePath, &sz)) {
                        LPCWSTR exeName = wcsrchr(exePath, L'\\');
                        exeName = exeName ? exeName + 1 : exePath;
                        g_targetIsExplorer = (_wcsicmp(exeName, L"explorer.exe") == 0);
                    }
                    CloseHandle(hProc);
                }
            }
        }
    }

    LoadMuiStrings();

    // Install hooks for DLLs already loaded
    HMODULE hWerui = GetModuleHandleW(L"werui.dll");
    if (hWerui) InstallWeruiDllHooks(hWerui);

    // Set up loading hooks if werui not present
    if (!hWerui) {
        Wh_SetFunctionHook((void*)LoadLibraryExW, (void*)LoadLibraryExW_hook,
                           (void**)&LoadLibraryExW_orig);
        Wh_SetFunctionHook((void*)LoadLibraryW, (void*)LoadLibraryW_hook,
                           (void**)&LoadLibraryW_orig);
#ifndef _WIN64
        Wh_SetFunctionHook((void*)LoadLibraryA, (void*)LoadLibraryA_hook,
                           (void**)&LoadLibraryA_orig);
        Wh_SetFunctionHook((void*)LoadLibraryExA, (void*)LoadLibraryExA_hook,
                           (void**)&LoadLibraryExA_orig);
#endif
        HMODULE hNt = GetModuleHandleW(L"ntdll.dll");
        if (hNt) {
            void* p = (void*)GetProcAddress(hNt, "LdrLoadDll");
            if (p) Wh_SetFunctionHook(p, (void*)LdrLoadDll_hook,
                                       (void**)&LdrLoadDll_orig);
        }
    }

    // Install wer.dll hooks
    HMODULE hWer = GetModuleHandleW(L"wer.dll");
    if (hWer) InstallWerDllHooks(hWer);

    // Apply everything so hooks are active before ReportProblem runs
    Wh_ApplyHookOperations();

    Wh_Log(L"Hooks installed (consent %d)", g_defaultConsent);
    return TRUE;
}
