/*

# Windows 7 AutoPlay Dialog



## Features

- Pixel-accurate Windows 7 AutoPlay dialog (grouped handler list, icon,

"Always do this" checkbox, Control Panel link)

- Full MTP/WPD support (phones, cameras, media players) via IPortableDeviceManager

- Volumes: USB flash drives, CD/DVD, external hard drives

- Autorun.inf parsing (shows program name + publisher from version info)

- Policy enforcement (NoDriveTypeAutoRun, NoDriveAutoRun, DisableAutoplay)

- IQueryCancelAutoPlay COM check (lets apps like WMP suppress the dialog)

- KnownDevices deduplication (no repeated dialogs for the same device)

- "Always do this" persists to HKCU UserChosenExecuteHandlers

- Classic theme and Aero theme both supported

- All standard Windows handler verbs supported (IHWEventHandler, ProgID+verb,

shell command)



## How it works

Windhawk injects this mod into explorer.exe. The mod registers a

CM_Register_Notification callback for device arrivals, intercepts them before

Explorer's own AutoPlay fires, and shows the Win7 dialog on a worker thread.

Explorer's own AutoPlay is suppressed via IQueryCancelAutoPlay registration

(the mod registers itself under CancelAutoplay\CLSID in HKCU).



## Notes

- Run Explorer as the injected process (default Windhawk target)

- The mod self-registers/unregisters IQueryCancelAutoPlay on load/unload

- First insertion after install may show both dialogs; subsequent ones will

only show the Win7 dialog

*/

// DispatchMessageW hook catches WM_DEVICECHANGE already flowing through

// explorer's own message loop (same technique as deviceinstallux)

using DispatchMessageW_t = decltype(&DispatchMessageW);

static DispatchMessageW_t orig_DispatchMessageW = nullptr;



static void SpawnVolumeDialog(LPCWSTR szPath, bool bFromDeviceChange = false); // forward decl

static void UnregisterKnownDevice(LPCWSTR szDevicePath); // forward decl

static void CloseDialogKey(const std::wstring& key); // forward decl

static void CloseAutoPlayDialogForDrive(WCHAR letter); // forward decl



static void TrySpawnVolumeDialogForMask(DWORD mask)

{

for (int i = 0; i < 26; i++)

{

if (!(mask & (1u << i))) continue;

WCHAR szDrive[4] = { (WCHAR)(L'A' + i), L':', L'\\', L'\0' };

Wh_Log(L"[DeviceArrival] Volume: %s", szDrive);

struct VA { std::wstring path; };

auto* va = new VA{ szDrive };

HANDLE h = CreateThread(nullptr, 0,

[](LPVOID pv) __attribute__((stdcall)) -> DWORD {

auto* va2 = reinterpret_cast<VA*>(pv);

HRESULT hr = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);

SpawnVolumeDialog(va2->path.c_str(), true);

if (SUCCEEDED(hr)) CoUninitialize();

delete va2; return 0;

}, va, 0, nullptr);

if (h) CloseHandle(h); else delete va;

}

}



LRESULT WINAPI hook_DispatchMessageW(const MSG* lpMsg)

{

if (lpMsg && lpMsg->message == WM_DEVICECHANGE)

{

auto* hdr = reinterpret_cast<DEV_BROADCAST_HDR*>(lpMsg->lParam);

if (lpMsg->wParam == DBT_DEVICEARRIVAL &&

hdr && hdr->dbch_devicetype == DBT_DEVTYP_VOLUME)

{

auto* vol = reinterpret_cast<DEV_BROADCAST_VOLUME*>(lpMsg->lParam);

TrySpawnVolumeDialogForMask(vol->dbcv_unitmask);

}

else if (lpMsg->wParam == DBT_DEVICEREMOVECOMPLETE &&

hdr && hdr->dbch_devicetype == DBT_DEVTYP_VOLUME)

{

auto* vol = reinterpret_cast<DEV_BROADCAST_VOLUME*>(lpMsg->lParam);

for (int i = 0; i < 26; i++)

{

if (!(vol->dbcv_unitmask & (1u << i))) continue;

WCHAR szDrive[4] = { (WCHAR)(L'A' + i), L':', L'\\', L'\0' };

CloseAutoPlayDialogForDrive(L'A' + i);

UnregisterKnownDevice(szDrive);

CloseDialogKey(std::wstring(szDrive));

}

}

}

return orig_DispatchMessageW ? orig_DispatchMessageW(lpMsg) : 0;

}

// Open dialog deduplication

static std::set<std::wstring> g_openDialogKeys;

static CRITICAL_SECTION g_csDialogKeys;

static bool g_csDialogKeysInit = false;



static void EnsureDialogCS()

{

if (!g_csDialogKeysInit)

{

InitializeCriticalSection(&g_csDialogKeys);

g_csDialogKeysInit = true;

}

}



static bool TryOpenDialog(const std::wstring& key)

{

EnsureDialogCS();

EnterCriticalSection(&g_csDialogKeys);

bool inserted = g_openDialogKeys.insert(key).second;

LeaveCriticalSection(&g_csDialogKeys);

return inserted;

}



static void CloseDialogKey(const std::wstring& key)

{

if (!g_csDialogKeysInit) return;

EnterCriticalSection(&g_csDialogKeys);

g_openDialogKeys.erase(key);

LeaveCriticalSection(&g_csDialogKeys);

}

static CRITICAL_SECTION g_csDialogs;

static std::map<WCHAR, HWND> g_openVolDialogs;

static std::map<std::wstring, HWND> g_openMTPDialogs;

static bool g_csDialogsInit = false;



static void EnsureDialogsCS()

{

if (!g_csDialogsInit)

{

InitializeCriticalSection(&g_csDialogs);

g_csDialogsInit = true;

}

}



static void RegisterVolDialog(WCHAR letter, HWND hwnd)

{

EnsureDialogsCS();

EnterCriticalSection(&g_csDialogs);

g_openVolDialogs[towupper(letter)] = hwnd;

LeaveCriticalSection(&g_csDialogs);

}



static void UnregisterVolDialog(WCHAR letter)

{

if (!g_csDialogsInit) return;

EnterCriticalSection(&g_csDialogs);

g_openVolDialogs.erase(towupper(letter));

LeaveCriticalSection(&g_csDialogs);

}



static void RegisterMTPDialog(const std::wstring& key, HWND hwnd)

{

EnsureDialogsCS();

EnterCriticalSection(&g_csDialogs);

g_openMTPDialogs[key] = hwnd;

LeaveCriticalSection(&g_csDialogs);

}



static void UnregisterMTPDialog(const std::wstring& key)

{

if (!g_csDialogsInit) return;

EnterCriticalSection(&g_csDialogs);

g_openMTPDialogs.erase(key);

LeaveCriticalSection(&g_csDialogs);

}



static void CloseAutoPlayDialogForDrive(WCHAR letter)

{

if (!g_csDialogsInit) return;

EnterCriticalSection(&g_csDialogs);

auto it = g_openVolDialogs.find(towupper(letter));

HWND hwnd = (it != g_openVolDialogs.end()) ? it->second : nullptr;

LeaveCriticalSection(&g_csDialogs);

if (hwnd && IsWindow(hwnd)) PostMessageW(hwnd, WM_CLOSE, 0, 0);

}



static void CloseAutoPlayDialogForDevice(LPCWSTR pszDeviceId)

{

if (!g_csDialogsInit || !pszDeviceId) return;

EnterCriticalSection(&g_csDialogs);

auto it = g_openMTPDialogs.find(pszDeviceId);

HWND hwnd = (it != g_openMTPDialogs.end()) ? it->second : nullptr;

LeaveCriticalSection(&g_csDialogs);

if (hwnd && IsWindow(hwnd)) PostMessageW(hwnd, WM_CLOSE, 0, 0);

}

static bool RegReadString(HKEY hKey, LPCWSTR pszValue, std::wstring& out)

{

DWORD dwType = 0, cbData = 0;

if (RegQueryValueExW(hKey, pszValue, nullptr, &dwType, nullptr, &cbData) != ERROR_SUCCESS)

return false;

if (cbData == 0) { out.clear(); return true; }

std::wstring buf(cbData / sizeof(WCHAR) + 2, L'\0');

if (RegQueryValueExW(hKey, pszValue, nullptr, &dwType,

reinterpret_cast<LPBYTE>(&buf[0]), &cbData) != ERROR_SUCCESS)

return false;

buf.resize(wcsnlen(buf.data(), buf.size()));

if (dwType == REG_EXPAND_SZ)

{

WCHAR exp[MAX_PATH * 2] = {};

ExpandEnvironmentStringsW(buf.c_str(), exp, ARRAYSIZE(exp));

out = exp;

}

else out = buf;

return true;

}



static std::wstring ResolveIndirect(const std::wstring& src)

{

if (src.empty() || src[0] != L'@') return src;

WCHAR buf[512] = {};

if (SUCCEEDED(SHLoadIndirectString(src.c_str(), buf, ARRAYSIZE(buf), nullptr)))

return buf;

return src;

}

static std::vector<std::wstring> ScanFilesForExtensions(

LPCWSTR pwszDevicePath, const std::vector<std::wstring>& exts)

{

std::vector<std::wstring> files;

if (!pwszDevicePath) return files;

for (auto& ext : exts)

{

WCHAR szSearch[MAX_PATH] = {};

StringCchPrintfW(szSearch, ARRAYSIZE(szSearch), L"%s*.%s", pwszDevicePath, ext.c_str());

WIN32_FIND_DATAW fd = {};

HANDLE h = FindFirstFileW(szSearch, &fd);

if (h == INVALID_HANDLE_VALUE) continue;

do {

if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) continue;

WCHAR szFull[MAX_PATH] = {};

StringCchPrintfW(szFull, ARRAYSIZE(szFull), L"%s%s", pwszDevicePath, fd.cFileName);

files.push_back(szFull);

} while (FindNextFileW(h, &fd));

FindClose(h);

}

return files;

}



static bool HasFiles(LPCWSTR pwszPath, LPCWSTR pwszPattern)

{

WCHAR szSearch[MAX_PATH] = {};

StringCchPrintfW(szSearch, ARRAYSIZE(szSearch), L"%s%s", pwszPath, pwszPattern);

WIN32_FIND_DATAW fd = {};

HANDLE h = FindFirstFileW(szSearch, &fd);

if (h != INVALID_HANDLE_VALUE) { FindClose(h); return true; }

WCHAR szDirs[MAX_PATH] = {};

StringCchPrintfW(szDirs, ARRAYSIZE(szDirs), L"%s*", pwszPath);

h = FindFirstFileW(szDirs, &fd);

if (h == INVALID_HANDLE_VALUE) return false;

do {

if (!(fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)) continue;

if (fd.cFileName[0] == L'.') continue;

WCHAR szSub[MAX_PATH] = {};

StringCchPrintfW(szSub, ARRAYSIZE(szSub), L"%s%s\\%s", pwszPath, fd.cFileName, pwszPattern);

WIN32_FIND_DATAW fd2 = {};

HANDLE h2 = FindFirstFileW(szSub, &fd2);

if (h2 != INVALID_HANDLE_VALUE) { FindClose(h2); FindClose(h); return true; }

} while (FindNextFileW(h, &fd));

FindClose(h);

return false;

}



static DWORD DetectContent(LPCWSTR szPath)

{

if (!szPath || !szPath[0]) return 0;

UINT t = GetDriveTypeW(szPath);

DWORD f = 0;

switch (t)

{

case DRIVE_CDROM:

if (HasFiles(szPath, L"autorun.inf")) f = AUTOPLAY_CONTENT_AUTORUN;

else if (HasFiles(szPath, L"*.mp3") || HasFiles(szPath, L"*.wav") ||

HasFiles(szPath, L"*.wma") || HasFiles(szPath, L"*.flac") ||

HasFiles(szPath, L"*.aac") || HasFiles(szPath, L"*.ogg")) f = AUTOPLAY_CONTENT_MUSIC;

else if (HasFiles(szPath, L"*.mp4") || HasFiles(szPath, L"*.avi") ||

HasFiles(szPath, L"*.mkv") || HasFiles(szPath, L"*.mov") ||

HasFiles(szPath, L"*.wmv") || HasFiles(szPath, L"*.vob")) f = AUTOPLAY_CONTENT_VIDEO;

else f = AUTOPLAY_CONTENT_BLANK_CD;

break;

case DRIVE_REMOVABLE: case DRIVE_FIXED:

if (HasFiles(szPath, L"autorun.inf") || HasFiles(szPath, L"setup.exe") ||

HasFiles(szPath, L"install.exe") || HasFiles(szPath, L"setup.msi"))

f |= AUTOPLAY_CONTENT_AUTORUN;

if (HasFiles(szPath, L"*.mp3") || HasFiles(szPath, L"*.flac") || HasFiles(szPath, L"*.wav") ||

HasFiles(szPath, L"*.wma") || HasFiles(szPath, L"*.aac") || HasFiles(szPath, L"*.ogg") ||

HasFiles(szPath, L"*.m4a") || HasFiles(szPath, L"*.opus"))

f |= AUTOPLAY_CONTENT_MUSIC;

if (HasFiles(szPath, L"*.jpg") || HasFiles(szPath, L"*.jpeg") || HasFiles(szPath, L"*.png") ||

HasFiles(szPath, L"*.bmp") || HasFiles(szPath, L"*.gif") || HasFiles(szPath, L"*.tiff") ||

HasFiles(szPath, L"*.webp") || HasFiles(szPath, L"*.heic") || HasFiles(szPath, L"*.cr2") ||

HasFiles(szPath, L"*.nef"))

f |= AUTOPLAY_CONTENT_PICTURES;

if (HasFiles(szPath, L"*.mp4") || HasFiles(szPath, L"*.avi") || HasFiles(szPath, L"*.mkv") ||

HasFiles(szPath, L"*.mov") || HasFiles(szPath, L"*.wmv") || HasFiles(szPath, L"*.m4v") ||

HasFiles(szPath, L"*.flv") || HasFiles(szPath, L"*.mpg") || HasFiles(szPath, L"*.mpeg") ||

HasFiles(szPath, L"*.ts") || HasFiles(szPath, L"*.vob") || HasFiles(szPath, L"*.webm"))

f |= AUTOPLAY_CONTENT_VIDEO;

if (f == 0) f = AUTOPLAY_CONTENT_OPENCPANEL;

break;

default:

f = AUTOPLAY_CONTENT_OPENCPANEL;

break;

}

return f;

}

static std::shared_ptr<HandlerEntry> LoadHandlerFromKey(

HKEY hRoot, LPCWSTR pwszName, LPCWSTR pwszEvent)

{

HKEY hSub = nullptr;

if (RegOpenKeyExW(hRoot, pwszName, 0, KEY_READ, &hSub) != ERROR_SUCCESS)

return nullptr;



auto e = std::make_shared<HandlerEntry>();

e->id = pwszName;

e->eventHandler = pwszEvent ? pwszEvent : L"";



std::wstring raw;

if (RegReadString(hSub, L"Action", raw)) e->action = ResolveIndirect(raw);

if (RegReadString(hSub, L"Provider", raw) && !raw.empty())

{

e->provider = ResolveIndirect(raw);

if (!e->provider.empty() && e->provider.find(L"using ") == std::wstring::npos)

e->provider = L"using " + e->provider;

}

if (RegReadString(hSub, L"DefaultIcon", raw)) e->iconLocation = raw;

if (!RegReadString(hSub, L"InvokeProgID", e->progId))

RegReadString(hSub, L"InvokeProgId", e->progId);

RegReadString(hSub, L"InvokeVerb", e->verb);

RegReadString(hSub, L"InitCmdLine", e->initCmdLine);



if (!e->progId.empty()) CLSIDFromProgID(e->progId.c_str(), &e->clsid);



std::wstring clsidStr;

if (RegReadString(hSub, L"CLSID", clsidStr) && !clsidStr.empty())

CLSIDFromString(clsidStr.c_str(), &e->clsid);



if (!e->progId.empty() && !e->verb.empty())

{

WCHAR cmd[MAX_PATH * 2] = {};

DWORD cch = ARRAYSIZE(cmd);

if (SUCCEEDED(AssocQueryStringW(0, ASSOCSTR_COMMAND,

e->progId.c_str(), e->verb.c_str(), cmd, &cch)))

e->command = cmd;

}



// Provider fallbacks

if (e->provider.empty() && !e->progId.empty())

{

WCHAR szFriendly[256] = {};

DWORD cch = ARRAYSIZE(szFriendly);

if (SUCCEEDED(AssocQueryStringW(0, ASSOCSTR_FRIENDLYAPPNAME,

e->progId.c_str(), nullptr, szFriendly, &cch)) && szFriendly[0])

{

e->provider = L"using ";

e->provider += szFriendly;

}

}

if (e->provider.empty() && !IsEqualCLSID(e->clsid, CLSID_NULL))

{

WCHAR szClsid[64] = {};

StringFromGUID2(e->clsid, szClsid, ARRAYSIZE(szClsid));

WCHAR szKey[256] = {};

StringCchPrintfW(szKey, ARRAYSIZE(szKey), L"CLSID\\%s", szClsid);

HKEY hk = nullptr;

if (RegOpenKeyExW(HKEY_CLASSES_ROOT, szKey, 0, KEY_READ, &hk) == ERROR_SUCCESS)

{

WCHAR szName[256] = {};

DWORD cb = sizeof(szName);

if (RegQueryValueExW(hk, L"", nullptr, nullptr,

reinterpret_cast<LPBYTE>(szName), &cb) == ERROR_SUCCESS && szName[0])

{

e->provider = L"using ";

e->provider += szName;

}

RegCloseKey(hk);

}

}

if (e->provider.empty() && !e->command.empty())

{

WCHAR szPath[MAX_PATH] = {};

StringCchCopyW(szPath, ARRAYSIZE(szPath), e->command.c_str());

PathUnquoteSpacesW(szPath);

PathStripPathW(szPath);

PathRemoveExtensionW(szPath);

if (szPath[0]) { e->provider = L"using "; e->provider += szPath; }

}



RegQueryInfoKeyW(hSub, nullptr, nullptr, nullptr, nullptr, nullptr, nullptr, nullptr,

nullptr, nullptr, nullptr, &e->lastWrite);

RegCloseKey(hSub);

return e;

}



static std::vector<std::shared_ptr<HandlerEntry>> LoadHandlersForEventHandler(

LPCWSTR pwszEventHandler, LPCWSTR /*pwszDevicePath*/, DWORD /*dwContentFlags*/)

{

std::vector<std::shared_ptr<HandlerEntry>> results;

WCHAR szEvKey[512] = {};

StringCchPrintfW(szEvKey, ARRAYSIZE(szEvKey),

L"%s%s", REGPATH_EVENTHANDLERS, pwszEventHandler);



const HKEY roots[2] = { HKEY_CURRENT_USER, HKEY_LOCAL_MACHINE };

for (int ri = 0; ri < 2; ++ri)

{

HKEY hEvKey = nullptr;

if (RegOpenKeyExW(roots[ri], szEvKey, 0, KEY_READ, &hEvKey) != ERROR_SUCCESS) continue;



DWORD idx = 0;

WCHAR szName[256] = {};

for (;;)

{

DWORD cch = ARRAYSIZE(szName);

LONG r = RegEnumValueW(hEvKey, idx++, szName, &cch, nullptr, nullptr, nullptr, nullptr);

if (r == ERROR_NO_MORE_ITEMS) break;

if (r != ERROR_SUCCESS || !szName[0]) continue;

if (!lstrcmpiW(szName, HANDLER_TAKE_NO_ACTION)) continue;

if (!lstrcmpiW(szName, HANDLER_PROMPT_EACH)) continue;



bool dup = false;

for (auto& x : results)

if (!lstrcmpiW(x->id.c_str(), szName)) { dup = true; break; }

if (dup) continue;



const HKEY handlerRoots[2] = { HKEY_CURRENT_USER, HKEY_LOCAL_MACHINE };

std::shared_ptr<HandlerEntry> entry;

for (int hr2 = 0; hr2 < 2 && !entry; ++hr2)

{

HKEY hH = nullptr;

if (RegOpenKeyExW(handlerRoots[hr2], REGPATH_HANDLERS, 0, KEY_READ, &hH) == ERROR_SUCCESS)

{

entry = LoadHandlerFromKey(hH, szName, pwszEventHandler);

RegCloseKey(hH);

}

}

if (entry && !entry->action.empty())

results.push_back(std::move(entry));

}

RegCloseKey(hEvKey);

}

return results;

}

static bool LoadHandlerIcon(HandlerEntry& entry, bool bSmall)

{

if (entry.iconLocation.empty()) return false;



WCHAR szIcon[MAX_PATH * 2] = {};

StringCchCopyW(szIcon, ARRAYSIZE(szIcon), entry.iconLocation.c_str());

WCHAR szExpanded[MAX_PATH * 2] = {};

ExpandEnvironmentStringsW(szIcon, szExpanded, ARRAYSIZE(szExpanded));

StringCchCopyW(szIcon, ARRAYSIZE(szIcon), szExpanded);



std::wstring iconStr = szIcon;

int idx = 0;

if (!iconStr.empty() && iconStr[0] == L'"')

{

size_t closeQ = iconStr.find(L'"', 1);

if (closeQ != std::wstring::npos)

{

size_t comma = iconStr.find(L',', closeQ);

if (comma != std::wstring::npos) idx = _wtoi(iconStr.c_str() + comma + 1);

iconStr = iconStr.substr(1, closeQ - 1);

}

}

else

{

size_t comma = iconStr.rfind(L',');

if (comma != std::wstring::npos)

{

idx = _wtoi(iconStr.c_str() + comma + 1);

iconStr = iconStr.substr(0, comma);

}

while (!iconStr.empty() && (iconStr.back() == L',' || iconStr.back() == L' '))

iconStr.pop_back();

}

StringCchCopyW(szIcon, ARRAYSIZE(szIcon), iconStr.c_str());



// Try system image list first (highest quality)

IImageList* pImgList = nullptr;

if (SUCCEEDED(SHGetImageList(SHIL_LARGE, IID_PPV_ARGS(&pImgList))))

{

SHFILEINFOW sfi2 = {};

SHGetFileInfoW(szIcon, 0, &sfi2, sizeof(sfi2), SHGFI_SYSICONINDEX | SHGFI_LARGEICON);

if (sfi2.iIcon >= 0) pImgList->GetIcon(sfi2.iIcon, ILD_TRANSPARENT, &entry.hIcon);

pImgList->Release();

if (entry.hIcon) return true;

}



// Fallback: ExtractIconExW

HICON hiconLarge = nullptr, hiconSmall = nullptr;

UINT nIcons = ExtractIconExW(szIcon, idx, &hiconLarge, &hiconSmall, 1);

if (nIcons == 0 && idx != 0)

nIcons = ExtractIconExW(szIcon, 0, &hiconLarge, &hiconSmall, 1);



if (!bSmall && hiconLarge)

{

entry.hIcon = hiconLarge;

if (hiconSmall) DestroyIcon(hiconSmall);

return true;

}

if (bSmall && hiconSmall)

{

entry.hIcon = hiconSmall;

if (hiconLarge) DestroyIcon(hiconLarge);

return true;

}

if (hiconLarge) DestroyIcon(hiconLarge);

if (hiconSmall) DestroyIcon(hiconSmall);

return false;

}

static bool AutoPlay_IsAllowedForDrive(LPCWSTR szRootPath)

{

if (!szRootPath || !szRootPath[0]) return false;



// DisableAutoplay

{

DWORD val = 0; HKEY hk = nullptr;

if (RegOpenKeyExW(HKEY_CURRENT_USER,

L"Software\\Microsoft\\Windows\\CurrentVersion\\Explorer\\AutoplayHandlers",

0, KEY_READ, &hk) == ERROR_SUCCESS)

{

DWORD cb = sizeof(val);

RegQueryValueExW(hk, L"DisableAutoplay", nullptr, nullptr,

reinterpret_cast<LPBYTE>(&val), &cb);

RegCloseKey(hk);

}

if (val) return false;

}



UINT driveType = GetDriveTypeW(szRootPath);



// NoDriveTypeAutoRun

{

DWORD dwMask = 0x91; // Win7 default (unknown + no-root + network + ramdisk)

for (HKEY root : { HKEY_LOCAL_MACHINE, HKEY_CURRENT_USER })

{

HKEY hk = nullptr;

if (RegOpenKeyExW(root,

L"Software\\Microsoft\\Windows\\CurrentVersion\\Policies\\Explorer",

0, KEY_READ, &hk) == ERROR_SUCCESS)

{

DWORD val = 0, cb = sizeof(val);

if (RegQueryValueExW(hk, L"NoDriveTypeAutoRun", nullptr, nullptr,

reinterpret_cast<LPBYTE>(&val), &cb) == ERROR_SUCCESS)

dwMask = val;

RegCloseKey(hk);

}

}

if (dwMask & (1u << driveType)) return false;

}



// NoDriveAutoRun (per drive letter)

if (szRootPath[1] == L':')

{

DWORD dwMask = 0;

for (HKEY root : { HKEY_LOCAL_MACHINE, HKEY_CURRENT_USER })

{

HKEY hk = nullptr;

if (RegOpenKeyExW(root,

L"Software\\Microsoft\\Windows\\CurrentVersion\\Policies\\Explorer",

0, KEY_READ, &hk) == ERROR_SUCCESS)

{

DWORD val = 0, cb = sizeof(val);

if (RegQueryValueExW(hk, L"NoDriveAutoRun", nullptr, nullptr,

reinterpret_cast<LPBYTE>(&val), &cb) == ERROR_SUCCESS)

dwMask = val;

RegCloseKey(hk);

}

}

if (dwMask)

{

int idx = towupper(szRootPath[0]) - L'A';

if (idx >= 0 && idx <= 25 && (dwMask & (1u << idx))) return false;

}

}



return true;

}

// ===========================================================================

// KnownDevices in-memory set, cleared on every mod load.

// Prevents showing the dialog twice for the same device in one session,

// but resets cleanly when the mod reloads so stale entries never block.

// ===========================================================================

static std::unordered_set<std::wstring> g_knownDevices;

static CRITICAL_SECTION g_csKnownDevices;



static std::wstring GetDeviceKey(LPCWSTR szDevicePath)

{

if (!szDevicePath || !szDevicePath[0]) return {};

// For drive letters, use the volume GUID so the key survives drive-letter changes

if (szDevicePath[1] == L':')

{

WCHAR szRoot[8] = {};

StringCchPrintfW(szRoot, ARRAYSIZE(szRoot), L"%c:\\", szDevicePath[0]);

WCHAR szGUID[MAX_PATH] = {};

if (GetVolumeNameForVolumeMountPointW(szRoot, szGUID, ARRAYSIZE(szGUID)) && szGUID[0])

return szGUID;

}

return szDevicePath;

}



static bool IsKnownDevice(LPCWSTR szDevicePath)

{

auto key = GetDeviceKey(szDevicePath);

if (key.empty()) return false;

EnterCriticalSection(&g_csKnownDevices);

bool known = g_knownDevices.count(key) > 0;

LeaveCriticalSection(&g_csKnownDevices);

Wh_Log(L"[IsKnownDevice] %s -> %s", szDevicePath, known ? L"known" : L"new");

return known;

}



static void RegisterKnownDevice(LPCWSTR szDevicePath)

{

auto key = GetDeviceKey(szDevicePath);

if (key.empty()) return;

EnterCriticalSection(&g_csKnownDevices);

g_knownDevices.insert(key);

LeaveCriticalSection(&g_csKnownDevices);

Wh_Log(L"[RegisterKnownDevice] %s", szDevicePath);

}



static void UnregisterKnownDevice(LPCWSTR szDevicePath)

{

auto key = GetDeviceKey(szDevicePath);

if (key.empty()) return;

EnterCriticalSection(&g_csKnownDevices);

g_knownDevices.erase(key);

LeaveCriticalSection(&g_csKnownDevices);

Wh_Log(L"[UnregisterKnownDevice] %s", szDevicePath);

}

static HRESULT InvokeHandler(const HandlerEntry& entry,

LPCWSTR pwszDevicePath, HWND hwndOwner)

{

// Case 1: COM IHWEventHandler

if (!IsEqualCLSID(entry.clsid, CLSID_NULL))

{

IHWEventHandler* pHW = nullptr;

HRESULT hr = CoCreateInstance(entry.clsid, nullptr,

CLSCTX_LOCAL_SERVER | CLSCTX_INPROC_SERVER,

IID_IHWEventHandler, reinterpret_cast<void**>(&pHW));

if (SUCCEEDED(hr) && pHW)

{

LPCWSTR pszInit = entry.initCmdLine.empty()

? pwszDevicePath : entry.initCmdLine.c_str();

hr = pHW->Initialize(pszInit);

if (SUCCEEDED(hr))

hr = pHW->HandleEvent(pwszDevicePath, pwszDevicePath,

entry.eventHandler.empty() ? L"" : entry.eventHandler.c_str());

pHW->Release();

if (SUCCEEDED(hr)) return hr;

}

}



// Case 2: Shell verb via ProgID

if (!entry.progId.empty() && !entry.verb.empty())

{

auto extensions = GetExtensionsForEvent(entry.eventHandler);

std::wstring targetPath = pwszDevicePath ? pwszDevicePath : L"";

if (!extensions.empty())

{

auto files = ScanFilesForExtensions(pwszDevicePath, extensions);

if (!files.empty()) targetPath = files[0];

}

IShellItem* psi = nullptr;

if (SUCCEEDED(SHCreateItemFromParsingName(targetPath.c_str(), nullptr, IID_PPV_ARGS(&psi))))

{

IContextMenu* pcm = nullptr;

if (SUCCEEDED(psi->BindToHandler(nullptr, BHID_SFUIObject, IID_PPV_ARGS(&pcm))))

{

char szVerb[64] = {};

WideCharToMultiByte(CP_ACP, 0, entry.verb.c_str(), -1,

szVerb, sizeof(szVerb), nullptr, nullptr);

CMINVOKECOMMANDINFO ici = {};

ici.cbSize = sizeof(ici);

ici.hwnd = hwndOwner;

ici.lpVerb = szVerb;

ici.nShow = SW_SHOWNORMAL;

HRESULT hr = pcm->InvokeCommand(&ici);

pcm->Release(); psi->Release();

if (SUCCEEDED(hr)) return hr;

}

else psi->Release();

}

SHELLEXECUTEINFOW sei = { sizeof(sei) };

sei.fMask = SEE_MASK_CLASSNAME | SEE_MASK_FLAG_NO_UI;

sei.hwnd = hwndOwner;

sei.lpClass = entry.progId.c_str();

sei.lpVerb = entry.verb.c_str();

sei.lpFile = targetPath.c_str();

sei.nShow = SW_SHOWNORMAL;

if (ShellExecuteExW(&sei)) return S_OK;

}



// Case 3: Direct command

if (!entry.command.empty())

{

std::wstring cmd = entry.command;

std::wstring device = pwszDevicePath ? pwszDevicePath : L"";

bool hasPlaceholder = (cmd.find(L"%1") != std::wstring::npos ||

cmd.find(L"%L") != std::wstring::npos ||

cmd.find(L"%V") != std::wstring::npos);



auto replaceAll = [](std::wstring& str, const std::wstring& from, const std::wstring& to)

{

size_t pos = 0;

while ((pos = str.find(from, pos)) != std::wstring::npos)

{ str.replace(pos, from.size(), to); pos += to.size(); }

};



if (hasPlaceholder)

{

auto extensions = GetExtensionsForEvent(entry.eventHandler);

if (!extensions.empty())

{

auto files = ScanFilesForExtensions(pwszDevicePath, extensions);

if (!files.empty())

{

std::wstring firstFile = L"\"" + files[0] + L"\"";

replaceAll(cmd, L"%1", firstFile);

replaceAll(cmd, L"%L", firstFile);

replaceAll(cmd, L"%V", firstFile);

}

else

{

replaceAll(cmd, L"%1", L"\"" + device + L"\"");

replaceAll(cmd, L"%L", L"\"" + device + L"\"");

replaceAll(cmd, L"%V", L"\"" + device + L"\"");

}

}

else

{

replaceAll(cmd, L"%1", L"\"" + device + L"\"");

replaceAll(cmd, L"%L", L"\"" + device + L"\"");

replaceAll(cmd, L"%V", L"\"" + device + L"\"");

}

}



SHELLEXECUTEINFOW sei = { sizeof(sei) };

sei.fMask = SEE_MASK_FLAG_NO_UI;

sei.hwnd = hwndOwner;

sei.lpVerb = L"open";

sei.lpFile = cmd.c_str();

sei.nShow = SW_SHOWNORMAL;

if (ShellExecuteExW(&sei)) return S_OK;

ShellExecuteW(hwndOwner, L"open", cmd.c_str(), nullptr, nullptr, SW_SHOWNORMAL);

return S_OK;

}



return E_FAIL;

}



// ===========================================================================

// Win7 AutoPlay Dialog built as a Win32 dialog from a runtime template

// (Windhawk mods can't have resource files, so we build the DLGTEMPLATE

// programmatically and use an in-memory dialog.)

// ===========================================================================



// Helper: align to DWORD

static void* AlignPtr(void* p) {

uintptr_t addr = reinterpret_cast<uintptr_t>(p);

if (addr & 3) addr = (addr + 3) & ~3;

return reinterpret_cast<void*>(addr);

}



// Build a DLGTEMPLATE in heap memory that matches the Win7 shell32 dialog 1121.

// Returns the buffer (caller must HeapFree) and its size.

static DLGTEMPLATE* BuildDialogTemplate(DWORD& outSize)

{

// We'll build the template into a fixed-size buffer.

// Win7 dialog: 212 x 300 DLUs, 10 controls.

const size_t BUF = 8192;

BYTE* buf = reinterpret_cast<BYTE*>(HeapAlloc(GetProcessHeap(), HEAP_ZERO_MEMORY, BUF));

if (!buf) return nullptr;



BYTE* p = buf;



// DLGTEMPLATEEX (extended dialog)

auto WriteWord = [&](WORD w) { *reinterpret_cast<WORD*>(p) = w; p += 2; };

auto WriteDword = [&](DWORD dw) { *reinterpret_cast<DWORD*>(p) = dw; p += 4; };

auto WriteWStr = [&](LPCWSTR s) {

while (*s) { WriteWord((WORD)*s++); }

WriteWord(0);

};

auto Align = [&]() {

while ((p - buf) & 3) { *p++ = 0; }

};



// DLGTEMPLATEEX header

WriteWord(1); // dlgVer

WriteWord(0xFFFF); // signature

WriteDword(0); // helpID

WriteDword(0); // exStyle

WriteDword(DS_SHELLFONT | DS_MODALFRAME | DS_CENTER | WS_MINIMIZEBOX |

WS_POPUP | WS_CLIPSIBLINGS | WS_CAPTION | WS_SYSMENU); // style matches RC

WriteWord(10); // cDlgItems

WriteWord(0); // x

WriteWord(0); // y

WriteWord(212); // cx (dialog units)

WriteWord(300); // cy

WriteWord(0); // menu (none)

WriteWord(0); // windowClass (none)

WriteWStr(L"AutoPlay"); // title

// DS_SHELLFONT font info

WriteWord(8); // pointsize

WriteWord(400); // weight (FW_NORMAL)

*p++ = 0; // bItalic

*p++ = 0; // charset ANSI_CHARSET = 0 (DEFAULT_CHARSET per DLGTEMPLATEEX)

WriteWStr(L"MS Shell Dlg");

Align();



// Helper to write a DLGITEMTEMPLATEEX

auto WriteControl = [&](DWORD exStyle, DWORD style,

short x, short y, short cx, short cy,

DWORD id, LPCWSTR cls, LPCWSTR title)

{

Align();

WriteDword(0); // helpID

WriteDword(exStyle); // exStyle

WriteDword(style); // style

WriteWord((WORD)x); WriteWord((WORD)y);

WriteWord((WORD)cx); WriteWord((WORD)cy);

static INT_PTR CALLBACK DlgProcStatic(HWND h, UINT m, WPARAM w, LPARAM l)

{

CAutoplayDialog* p = nullptr;

if (m == WM_INITDIALOG)

{

p = reinterpret_cast<CAutoplayDialog*>(l);

Wh_Log(L"[DlgProc] WM_INITDIALOG hwnd=%p p=%p", h, p);

SetWindowLongPtrW(h, DWLP_USER, reinterpret_cast<LONG_PTR>(p));

p->m_hwnd = h;

}

else p = reinterpret_cast<CAutoplayDialog*>(GetWindowLongPtrW(h, DWLP_USER));

return p ? p->Proc(h, m, w, l) : FALSE;

}



private:

INT_PTR Proc(HWND h, UINT msg, WPARAM w, LPARAM l)

{

switch (msg)

{

case WM_INITDIALOG: return Init();



case WM_COMMAND:

if (LOWORD(w) == IDOK) { OnOK(); return TRUE; }

if (LOWORD(w) == IDCANCEL) { EndDialog(h, IDCANCEL); return TRUE; }

if (LOWORD(w) == IDC_AUTOPLAY_ALWAYSDO && HIWORD(w) == BN_CLICKED)

{

m_bAlwaysDo = (SendDlgItemMessageW(h, IDC_AUTOPLAY_ALWAYSDO,

BM_GETCHECK, 0, 0) == BST_CHECKED);

return TRUE;

}

break;



case WM_NOTIFY:

{

auto* nm = reinterpret_cast<NMHDR*>(l);

if (nm->idFrom == IDC_AUTOPLAY_LIST)

{

if (nm->code == NM_CUSTOMDRAW)

{

LRESULT lr = CustomDraw(reinterpret_cast<NMLVCUSTOMDRAW*>(l));

SetWindowLongPtrW(m_hwnd, DWLP_MSGRESULT, lr);

return TRUE;

}

if (nm->code == NM_DBLCLK || nm->code == LVN_ITEMACTIVATE)

{ OnOK(); return TRUE; }

}

else if (nm->idFrom == IDC_AUTOPLAY_CPLLINK &&

(nm->code == NM_CLICK || nm->code == NM_RETURN))

OpenControlPanelAutoPlay();

break;

}



case WM_CTLCOLORSTATIC:

return ColorStatic(reinterpret_cast<HDC>(w), reinterpret_cast<HWND>(l));



case WM_ERASEBKGND:

if (m_hbrBg)

{

RECT r2; GetClientRect(h, &r2);

FillRect((HDC)w, &r2, m_hbrBg);

return TRUE;

}

break;



case WM_CTLCOLORBTN:

{

HWND hwndCtl = reinterpret_cast<HWND>(l);

if (GetDlgCtrlID(hwndCtl) == IDC_AUTOPLAY_ALWAYSDO)

{

HDC hdc = reinterpret_cast<HDC>(w);

SetBkMode(hdc, TRANSPARENT);

if (m_hbrBg) return reinterpret_cast<INT_PTR>(m_hbrBg);

}

break;

}



case WM_DESTROY:

if (m_forDevice && !m_devicePath.empty())

UnregisterMTPDialog(m_devicePath);

else if (!m_devicePath.empty() && m_devicePath.size() <= 3 && m_devicePath[1] == L':')

{

WCHAR letter = towupper(m_devicePath[0]);

UnregisterVolDialog(letter);

}

FreeTheme();

ClearList();

break;



case WM_TIMER:

if (w == 1) { KillTimer(h, 1); if (m_state == APState::Searching) GoChooser(); }

break;



case WM_CLOSE:

EndDialog(h, IDCANCEL);

return TRUE;



case WM_AUTOPLAY_NEWCONTENT: m_contentFlags |= (DWORD)l; return TRUE;

case WM_AUTOPLAY_SNIFFCOMPLETE:

KillTimer(h, 1);

m_contentFlags = (DWORD)l;

GoChooser();

return TRUE;

}

return FALSE;

}

LRESULT CustomDraw(NMLVCUSTOMDRAW* pcd)

{

switch (pcd->nmcd.dwDrawStage)

{

case CDDS_PREPAINT: return CDRF_NOTIFYITEMDRAW;

case CDDS_ITEMPREPAINT:

{

int idx = (int)pcd->nmcd.dwItemSpec;

if (idx < 0 || idx >= (int)m_items.size()) return CDRF_DODEFAULT;



HDC hdc = pcd->nmcd.hdc;

RECT rc = {};

ListView_GetItemRect(m_hwndList, idx, &rc, LVIR_BOUNDS);

RECT rcClient; GetClientRect(m_hwndList, &rcClient);

rc.right = rcClient.right;



bool bSel = (ListView_GetItemState(m_hwndList, idx, LVIS_SELECTED) & LVIS_SELECTED) != 0;

bool bHot = (ListView_GetHotItem(m_hwndList) == idx);

bool bThemed = IsAppThemed() && IsThemeActive();



// Background

if (bSel || bHot)

{

if (bThemed)

{

HTHEME hTh = OpenThemeData(m_hwndList, L"ListView");

if (hTh)

{

DrawThemeBackground(hTh, hdc, LVP_LISTITEM,

bSel ? LISS_SELECTED : LISS_HOT, &rc, nullptr);

CloseThemeData(hTh);

}

else

{

HBRUSH hbr = CreateSolidBrush(GetSysColor(COLOR_HIGHLIGHT));

FillRect(hdc, &rc, hbr); DeleteObject(hbr);

}

}

else

{

HBRUSH hbr = CreateSolidBrush(GetSysColor(COLOR_HIGHLIGHT));

FillRect(hdc, &rc, hbr); DeleteObject(hbr);

if (bSel)

{

HBRUSH hbrFrame = (HBRUSH)GetStockObject(NULL_BRUSH);

HBRUSH hbrOld = (HBRUSH)SelectObject(hdc, hbrFrame);

HPEN hpen = CreatePen(PS_DOT, 1, GetSysColor(COLOR_HIGHLIGHTTEXT));

HPEN hpenOld = (HPEN)SelectObject(hdc, hpen);

SetBkColor(hdc, GetSysColor(COLOR_HIGHLIGHT));

Rectangle(hdc, rc.left, rc.top, rc.right, rc.bottom);

SelectObject(hdc, hpenOld); DeleteObject(hpen);

SelectObject(hdc, hbrOld);

}

}

}

else

{

HBRUSH hbr = CreateSolidBrush(GetSysColor(COLOR_WINDOW));

FillRect(hdc, &rc, hbr); DeleteObject(hbr);

}

SetBkMode(hdc, TRANSPARENT);



const HandlerEntry* e = m_items[idx].entry.get();

if (!e) return CDRF_SKIPDEFAULT;



int rowH = rc.bottom - rc.top;

int midY = rc.top + rowH / 2;



if (e->hIcon)

DrawIconEx(hdc, rc.left + 4, rc.top + (rowH - 32) / 2,

e->hIcon, 32, 32, 0, nullptr, DI_NORMAL);



int textLeft = e->hIcon ? rc.left + 44 : rc.left + 6;



// Text colours

COLORREF clrAction, clrProvider;

if (bSel)

{

clrAction = GetSysColor(COLOR_HIGHLIGHTTEXT);

if (bThemed)

{

HTHEME hTh = OpenThemeData(m_hwndList, L"ListView");

if (hTh)

{

GetThemeColor(hTh, LVP_LISTITEM, LISS_SELECTED, TMT_TEXTCOLOR, &clrAction);

CloseThemeData(hTh);

}

}

clrProvider = clrAction;

}

else

{

clrAction = GetSysColor(COLOR_WINDOWTEXT);

clrProvider = GetSysColor(COLOR_GRAYTEXT);

}



LOGFONTW lf = {};

lf.lfHeight = -MulDiv(8, GetDeviceCaps(hdc, LOGPIXELSY), 72);

lf.lfWeight = FW_NORMAL;

lf.lfUnderline = (bThemed && (bSel || bHot)) ? TRUE : FALSE;

lf.lfCharSet = DEFAULT_CHARSET;

lf.lfQuality = bThemed ? CLEARTYPE_QUALITY : DEFAULT_QUALITY;

lf.lfPitchAndFamily = DEFAULT_PITCH | FF_SWISS;

StringCchCopyW(lf.lfFaceName, LF_FACESIZE, L"Tahoma");



HFONT hfN = CreateFontIndirectW(&lf);

HFONT hfO = (HFONT)SelectObject(hdc, hfN);

SetTextColor(hdc, clrAction);

RECT rcA = { textLeft, rc.top + 2, rc.right - 4, midY };

DrawTextW(hdc, e->action.c_str(), -1, &rcA,

DT_LEFT | DT_BOTTOM | DT_SINGLELINE | DT_END_ELLIPSIS);

SelectObject(hdc, hfO); DeleteObject(hfN);



if (!e->provider.empty())

{

HFONT hfP = CreateFontIndirectW(&lf);

hfO = (HFONT)SelectObject(hdc, hfP);

SetTextColor(hdc, clrProvider);

RECT rcP = { textLeft, midY, rc.right - 4, rc.bottom - 2 };

DrawTextW(hdc, e->provider.c_str(), -1, &rcP,

DT_LEFT | DT_TOP | DT_SINGLELINE | DT_END_ELLIPSIS);

SelectObject(hdc, hfO); DeleteObject(hfP);

}

return CDRF_SKIPDEFAULT;

}

}

return CDRF_DODEFAULT;

}

void OnOK()

{

int i = ListView_GetNextItem(m_hwndList, -1, LVNI_SELECTED);

if (i < 0 || i >= (int)m_items.size()) { EndDialog(m_hwnd, IDCANCEL); return; }

auto* e = m_items[i].entry.get();

if (!e) { EndDialog(m_hwnd, IDCANCEL); return; }



if (e->id == HANDLER_TAKE_NO_ACTION)

{

if (m_bAlwaysDo && !m_eventHandler.empty())

SetDefaultHandler(m_eventHandler.c_str(), HANDLER_TAKE_NO_ACTION);

EndDialog(m_hwnd, IDCANCEL); return;

}

if (e->id == HANDLER_OPEN_FOLDER)

{

ShellExecuteW(m_hwnd, L"open", m_devicePath.c_str(), nullptr, nullptr, SW_SHOWNORMAL);

if (m_bAlwaysDo && !m_eventHandler.empty())

SetDefaultHandler(m_eventHandler.c_str(), HANDLER_OPEN_FOLDER);

EndDialog(m_hwnd, IDOK); return;

}

if (e->id == L"MSAutoRun")

{

WCHAR szInf[MAX_PATH] = {}, szOpen[MAX_PATH] = {};

StringCchPrintfW(szInf, ARRAYSIZE(szInf), L"%sautorun.inf", m_devicePath.c_str());

GetPrivateProfileStringW(L"AutoRun.Amd64", L"open", L"", szOpen, ARRAYSIZE(szOpen), szInf);

if (!szOpen[0]) GetPrivateProfileStringW(L"AutoRun", L"open", L"", szOpen, ARRAYSIZE(szOpen), szInf);

if (szOpen[0])

{

WCHAR szExeFull[MAX_PATH] = {};

StringCchPrintfW(szExeFull, ARRAYSIZE(szExeFull), L"%s%s", m_devicePath.c_str(), szOpen);

SHELLEXECUTEINFOW sei = { sizeof(sei) };

sei.fMask = SEE_MASK_FLAG_NO_UI; sei.hwnd = m_hwnd;

sei.lpVerb = L"open"; sei.lpFile = szExeFull; sei.nShow = SW_SHOWNORMAL;

ShellExecuteExW(&sei);

}

if (m_bAlwaysDo && !m_eventHandler.empty())

SetDefaultHandler(m_eventHandler.c_str(), e->id.c_str());

EndDialog(m_hwnd, IDOK); return;

}

if (e->id == L"MSWPDShellNamespaceHandler")

{

static const CLSID CLSID_WPDShell =

{ 0xA55803CC, 0x4D53, 0x404c, {0x85,0x57,0xFD,0x63,0xDB,0xA9,0x5D,0x24} };

IHWEventHandler* pHW = nullptr;

if (SUCCEEDED(CoCreateInstance(CLSID_WPDShell, nullptr,

CLSCTX_LOCAL_SERVER | CLSCTX_INPROC_SERVER,

IID_IHWEventHandler, reinterpret_cast<void**>(&pHW))) && pHW)

{

if (SUCCEEDED(pHW->Initialize(m_devicePath.c_str())))

pHW->HandleEvent(m_devicePath.c_str(), m_devicePath.c_str(),

m_eventHandler.c_str());

pHW->Release();

}

if (m_bAlwaysDo && !m_eventHandler.empty())

SetDefaultHandler(m_eventHandler.c_str(), e->id.c_str());

EndDialog(m_hwnd, IDOK); return;

}

if (m_bAlwaysDo && !m_eventHandler.empty())

SetDefaultHandler(m_eventHandler.c_str(), e->id.c_str());

ShowWindow(m_hwnd, SW_HIDE);

InvokeHandler(*e, m_devicePath.c_str(), nullptr);

EndDialog(m_hwnd, IDOK);

}

void GoChooser()

{

m_state = APState::Chooser;

Vis(IDC_AUTOPLAY_SEARCHING, false); Vis(IDC_AUTOPLAY_PROGRESS, false);

Vis(IDC_AUTOPLAY_DEFAULT_MSG, false); Vis(IDC_AUTOPLAY_NEWHANDLER, false);

Vis(IDC_AUTOPLAY_TITLE, true); Vis(IDC_AUTOPLAY_CPLLINK, true);



Populate();

Vis(IDC_AUTOPLAY_LIST, true);



int nContentTypes = 0;

DWORD t2 = m_contentFlags & (AUTOPLAY_CONTENT_MUSIC | AUTOPLAY_CONTENT_PICTURES |

AUTOPLAY_CONTENT_VIDEO | AUTOPLAY_CONTENT_AUTORUN);

while (t2) { nContentTypes += (t2 & 1); t2 >>= 1; }

bool bMixed = (nContentTypes > 1);

bool bIsFixed = false;

if (!m_devicePath.empty())

bIsFixed = (GetDriveTypeW(m_devicePath.c_str()) == DRIVE_FIXED);



std::wstring chkText = LoadStr(IDS_ALWAYS_DO_DEVICE);

bool showChk = true;

if (bMixed) showChk = false;

else if (m_contentFlags & AUTOPLAY_CONTENT_AUTORUN) chkText = LoadStr(IDS_ALWAYS_DO_AUTORUN);

else if (m_contentFlags & AUTOPLAY_CONTENT_MUSIC) chkText = LoadStr(IDS_ALWAYS_DO_MUSIC);

else if (m_contentFlags & AUTOPLAY_CONTENT_PICTURES) chkText = LoadStr(IDS_ALWAYS_DO_PICTURES);

else if (m_contentFlags & AUTOPLAY_CONTENT_VIDEO) chkText = LoadStr(IDS_ALWAYS_DO_VIDEO);

else if (m_contentFlags & AUTOPLAY_CONTENT_BLANK_CD) chkText = LoadStr(IDS_ALWAYS_DO_BLANK_CD);

else if (m_contentFlags & AUTOPLAY_CONTENT_BLANK_DVD) chkText = LoadStr(IDS_ALWAYS_DO_BLANK_DVD);

else showChk = false;



SetDlgItemTextW(m_hwnd, IDC_AUTOPLAY_ALWAYSDO, chkText.c_str());

HWND hChk = GetDlgItem(m_hwnd, IDC_AUTOPLAY_ALWAYSDO);

if (showChk && !bIsFixed)

{

bool bChk = !m_defaultHandlerID.empty() &&

m_defaultHandlerID != HANDLER_PROMPT_EACH &&

m_defaultHandlerID != HANDLER_TAKE_NO_ACTION;

SetWindowTheme(hChk, L"Button", nullptr);

ShowWindow(hChk, SW_SHOW);

EnableWindow(hChk, TRUE);

SendMessageW(hChk, BM_SETCHECK, bChk ? BST_CHECKED : BST_UNCHECKED, 0);

m_bAlwaysDo = bChk;

}

else { ShowWindow(hChk, SW_HIDE); m_bAlwaysDo = false; }



Select(m_defaultHandlerID.empty() ? nullptr : m_defaultHandlerID.c_str());

Resize(showChk);

ShowWindow(m_hwnd, SW_SHOW);

SetForegroundWindow(m_hwnd);

}

void Populate()

{

SendMessageW(m_hwndList, WM_SETREDRAW, FALSE, 0);

ClearList();

bool addedAnyGroup = false;



int nContentTypes = 0;

DWORD t2 = m_contentFlags & (AUTOPLAY_CONTENT_MUSIC | AUTOPLAY_CONTENT_PICTURES |

AUTOPLAY_CONTENT_VIDEO | AUTOPLAY_CONTENT_AUTORUN);

while (t2) { nContentTypes += (t2 & 1); t2 >>= 1; }

bool bMixed = (nContentTypes > 1);



std::vector<std::pair<std::wstring, std::wstring>> groups;

std::vector<std::wstring> usedKeys;



if (m_forDevice)

{

// MTP/WPD device: interface path, not a drive letter

bool isMTPPath = m_devicePath.size() > 3 && m_devicePath[1] != L':';

if (isMTPPath)

{

AddGroup(GetStr(IDS_GROUP_DEVICE));

addedAnyGroup = true;



// Load MTP media player handlers

auto hlist = LoadHandlersForEventHandler(

L"MTPMediaPlayerArrival", m_devicePath.c_str(), m_contentFlags);



// Merge extra event handler handlers

if (m_eventHandler != L"MTPMediaPlayerArrival" && !m_eventHandler.empty())

{

// -----------------------------------------------------------------------

// Sniff thread (content detection for volumes)

// -----------------------------------------------------------------------

struct SP { CAutoplayDialog* p; HANDLE hReady; };



static unsigned __stdcall SniffProc(void* arg)

{

auto* sp = reinterpret_cast<SP*>(arg);

auto* p = sp->p;

HANDLE hR = sp->hReady;

delete sp;

if (hR) WaitForSingleObject(hR, 10000);

DWORD flags = DetectContent(p->m_devicePath.c_str());

if (p->m_hwnd) PostMessageW(p->m_hwnd, WM_AUTOPLAY_NEWCONTENT, 0, (LPARAM)flags);

Sleep(50);

if (p->m_hwnd) PostMessageW(p->m_hwnd, WM_AUTOPLAY_SNIFFCOMPLETE, 0, (LPARAM)flags);

return 0;

}



void Sniff()

{

auto* sp = new SP{ this, m_hReady };

HANDLE h = (HANDLE)_beginthreadex(nullptr, 0, SniffProc, sp, 0, nullptr);

if (h) CloseHandle(h); else delete sp;

}

};

bool bIsVolume)

{

const HRESULT hrCo = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);



CAutoplayDialog dlg;

dlg.m_devicePath = pwszDevicePath ? pwszDevicePath : L"";

dlg.m_volumeLabel = pwszLabel ? pwszLabel : L"";

}



// ===========================================================================

// SpawnParams / SpawnThread

// ===========================================================================

struct SpawnParams

{

std::wstring key;

std::wstring label;

std::wstring eventHandler;

DWORD contentFlags;

bool isVolume;

};



static DWORD WINAPI SpawnThread(LPVOID pv)

{

auto* p = reinterpret_cast<SpawnParams*>(pv);

const HRESULT hrCo = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);



if (p->isVolume && p->contentFlags == 0)

p->contentFlags = DetectContent(p->key.c_str());



ShowAutoPlayDialog(nullptr,

p->key.c_str(),

p->label.c_str(),

nullptr,

p->eventHandler.empty() ? nullptr : p->eventHandler.c_str(),

p->contentFlags,

p->isVolume);



CloseDialogKey(p->key);

RegisterKnownDevice(p->key.c_str());



if (SUCCEEDED(hrCo)) CoUninitialize();

delete p;

return 0;

}



static std::wstring GetDriveLabel(LPCWSTR szPath)

{

WCHAR vol[MAX_PATH] = {};

GetVolumeInformationW(szPath, vol, MAX_PATH, nullptr, nullptr, nullptr, nullptr, 0);

if (vol[0])

{

WCHAR buf[256] = {};

StringCchPrintfW(buf, ARRAYSIZE(buf), L"%s (%c:)", vol, szPath[0]);

return buf;

}

return LoadStrFmt(IDS_REMOVABLE_DISK, szPath[0]);

}



static void SpawnVolumeDialog(LPCWSTR szPath, bool bFromDeviceChange)

{

Wh_Log(L"[SpawnVolumeDialog] path=%s fromDC=%d", szPath ? szPath : L"(null)", bFromDeviceChange);

if (!szPath || !szPath[0]) { Wh_Log(L"[SpawnVolumeDialog] null/empty path"); return; }

UINT t = GetDriveTypeW(szPath);

Wh_Log(L"[SpawnVolumeDialog] DriveType=%u", t);

if (t == DRIVE_NO_ROOT_DIR || t == DRIVE_UNKNOWN || t == DRIVE_REMOTE) { Wh_Log(L"[SpawnVolumeDialog] rejected: bad drive type"); return; }

if (t != DRIVE_REMOVABLE && t != DRIVE_CDROM && t != DRIVE_FIXED) { Wh_Log(L"[SpawnVolumeDialog] rejected: not removable/cdrom/fixed"); return; }

if (t == DRIVE_FIXED && !bFromDeviceChange) { Wh_Log(L"[SpawnVolumeDialog] rejected: fixed drive not from device change"); return; }



if (!AutoPlay_IsAllowedForDrive(szPath)) { Wh_Log(L"[SpawnVolumeDialog] rejected: AutoPlay not allowed"); return; }

if (IsKnownDevice(szPath)) { Wh_Log(L"[SpawnVolumeDialog] rejected: already known device"); return; }



std::wstring key = szPath;

if (!TryOpenDialog(key)) { Wh_Log(L"[SpawnVolumeDialog] rejected: TryOpenDialog failed"); return; }

Wh_Log(L"[SpawnVolumeDialog] spawning thread for %s", szPath);



std::wstring label = GetDriveLabel(szPath);

auto* prm = new SpawnParams{ key, label, L"", 0, true };

HANDLE h = CreateThread(nullptr, 0, SpawnThread, prm, 0, nullptr);

if (h) CloseHandle(h);

else { CloseDialogKey(key); delete prm; }

}



static void SpawnMTPDialog(const std::wstring& devId, const std::wstring& label)

{

if (IsKnownDevice(devId.c_str())) return;

if (!TryOpenDialog(devId)) return;



// Use ShowPicturesOnArrival as default dialog shows all WPD handlers

auto* prm = new SpawnParams{ devId, label, L"ShowPicturesOnArrival", 0, false };

HANDLE h = CreateThread(nullptr, 0, SpawnThread, prm, 0, nullptr);

if (h) CloseHandle(h);

else { CloseDialogKey(devId); delete prm; }

}

// ===========================================================================

// Volume arrival: resolve interface path -> drive letter (with retry)

// ===========================================================================

static void OnVolumeArrived(const std::wstring& symLink)

{

Wh_Log(L"[OnVolumeArrived] symLink=%s", symLink.c_str());

WCHAR szDrive[8] = {};

for (int attempt = 0; attempt < 20 && !szDrive[0]; ++attempt)

{

if (attempt) Sleep(250);

WCHAR drives[512] = {};

GetLogicalDriveStringsW(ARRAYSIZE(drives) - 1, drives);

for (WCHAR* p = drives; *p; p += wcslen(p) + 1)

{

WCHAR szNT[MAX_PATH] = {};

WCHAR szLetter[3] = { p[0], L':', L'\0' };

if (!QueryDosDeviceW(szLetter, szNT, ARRAYSIZE(szNT))) continue;



// CM symlinks arrive as \\?\Volume{...} or \\?\STORAGE#Volume#...

// QueryDosDevice returns \Device\HarddiskVolume...

// Try direct compare first, then strip the \\?\ prefix and prepend \Device

std::wstring ntLink = symLink;

if (ntLink.size() > 4 &&

(ntLink.substr(0, 4) == L"\\\\?\\" || ntLink.substr(0, 4) == L"\\\\.\\"))

{

ntLink = ntLink.substr(4); // strip \\?\ or \\\. (prefix removed)

}

// Also try with \Device\ prefix since QueryDosDevice returns that form

std::wstring withDevice = L"\\Device\\" + ntLink;



if (_wcsicmp(szNT, symLink.c_str()) == 0 ||

_wcsicmp(szNT, ntLink.c_str()) == 0 ||

_wcsicmp(szNT, withDevice.c_str()) == 0)

{

StringCchPrintfW(szDrive, ARRAYSIZE(szDrive), L"%c:\\", p[0]);

Wh_Log(L"[OnVolumeArrived] matched drive %s (szNT=%s)", szDrive, szNT);

break;

}

}

}

if (!szDrive[0])

{

// Last resort: if a new drive letter appeared since plugin loaded,

// pick the first removable drive not previously seen.

Wh_Log(L"[OnVolumeArrived] symlink match failed, trying removable drive scan");

WCHAR drives[512] = {};

GetLogicalDriveStringsW(ARRAYSIZE(drives) - 1, drives);

for (WCHAR* p = drives; *p; p += wcslen(p) + 1)

{

if (GetDriveTypeW(p) == DRIVE_REMOVABLE && !IsKnownDevice(p))

{

StringCchCopyW(szDrive, ARRAYSIZE(szDrive), p);

Wh_Log(L"[OnVolumeArrived] fallback: using removable drive %s", szDrive);

break;

}

}

}

if (!szDrive[0]) { Wh_Log(L"[OnVolumeArrived] could not resolve drive, giving up"); return; }



UINT driveType = GetDriveTypeW(szDrive);

if (driveType == DRIVE_REMOTE || driveType == DRIVE_NO_ROOT_DIR) return;

// Skip system drive (C:) unless it's removable or CD

if (driveType == DRIVE_FIXED && towupper(szDrive[0]) == L'C') return;



SpawnVolumeDialog(szDrive, true);

}



static void OnVolumeRemoved(const std::wstring& symLink)

{

WCHAR drives[512] = {};

GetLogicalDriveStringsW(ARRAYSIZE(drives) - 1, drives);

for (WCHAR* p = drives; *p; p += wcslen(p) + 1)

{

WCHAR szNT[MAX_PATH] = {};

WCHAR szLetter[3] = { p[0], L':', L'\0' };

if (!QueryDosDeviceW(szLetter, szNT, ARRAYSIZE(szNT))) continue;



std::wstring ntLink;

if (symLink.size() > 4 && symLink.substr(0, 4) == L"\\\\?\\")

ntLink = L"\\Device\\" + symLink.substr(4);

else

ntLink = symLink;



if (_wcsicmp(szNT, ntLink.c_str()) == 0)

{

WCHAR szKey[8] = {};

StringCchPrintfW(szKey, ARRAYSIZE(szKey), L"%c:\\", p[0]);

CloseAutoPlayDialogForDrive(p[0]);

CloseDialogKey(szKey);

UnregisterKnownDevice(szKey);

break;

}

}

}



static void OnWPDArrived(const std::wstring& interfacePath)

{

Sleep(400); // Let WPD stack settle



IPortableDeviceManager* pMgr = nullptr;

if (FAILED(CoCreateInstance(CLSID_PortableDeviceManager, nullptr,

CLSCTX_INPROC_SERVER, IID_PPV_ARGS(&pMgr))))

return;



DWORD dwCount = 0;

pMgr->GetDevices(nullptr, &dwCount);

if (dwCount == 0) { pMgr->Release(); return; }



std::vector<LPWSTR> ids(dwCount, nullptr);

pMgr->GetDevices(ids.data(), &dwCount);



for (DWORD i = 0; i < dwCount; ++i)

{

if (!ids[i]) continue;

std::wstring devId = ids[i];

CoTaskMemFree(ids[i]);



if (!TryOpenDialog(devId)) continue;



WCHAR szName[256] = {};

DWORD cch = ARRAYSIZE(szName);

pMgr->GetDeviceFriendlyName(devId.c_str(), szName, &cch);



std::wstring label = szName[0] ? szName : LoadStr(IDS_UNKNOWN_DEVICE);



// Put back the key so SpawnMTPDialog can re-acquire it

CloseDialogKey(devId);

SpawnMTPDialog(devId, label);

}

pMgr->Release();

}



// ===========================================================================

// Message window proc (runs on Explorer's main thread)

// ===========================================================================

struct DevArrival

{

std::wstring interfacePath;

bool isVolume;

bool isWPD;

};



static LRESULT CALLBACK MsgWndProc(HWND hwnd, UINT msg, WPARAM w, LPARAM l)

{

switch (msg)

{

case WM_AP_ARRIVED:

{

auto* ev = reinterpret_cast<DevArrival*>(l);

if (!ev) return 0;

if (ev->isVolume)

{

struct VA { std::wstring path; };

auto* va = new VA{ ev->interfacePath };

HANDLE h = CreateThread(nullptr, 0, [](LPVOID pv) __attribute__((stdcall)) -> DWORD {

auto* va2 = reinterpret_cast<VA*>(pv);

HRESULT hr = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);

OnVolumeArrived(va2->path);

if (SUCCEEDED(hr)) CoUninitialize();

delete va2; return 0;

}, va, 0, nullptr);

if (h) CloseHandle(h); else delete va;

}

else if (ev->isWPD)

{

struct WA { std::wstring path; };

auto* wa = new WA{ ev->interfacePath };

HANDLE h = CreateThread(nullptr, 0, [](LPVOID pv) __attribute__((stdcall)) -> DWORD {

auto* wa2 = reinterpret_cast<WA*>(pv);

HRESULT hr = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);

OnWPDArrived(wa2->path);

if (SUCCEEDED(hr)) CoUninitialize();

delete wa2; return 0;

}, wa, 0, nullptr);

if (h) CloseHandle(h); else delete wa;

}

delete ev;

return 0;

}



case WM_AP_REMOVED:

{

auto* ev = reinterpret_cast<DevArrival*>(l);

if (!ev) return 0;

if (ev->isVolume)

OnVolumeRemoved(ev->interfacePath);

else

{

CloseAutoPlayDialogForDevice(ev->interfacePath.c_str());

CloseDialogKey(ev->interfacePath);

UnregisterKnownDevice(ev->interfacePath.c_str());

auto it = g_deviceIcons.find(ev->interfacePath);

if (it != g_deviceIcons.end())

{

if (it->second) DestroyIcon(it->second);

g_deviceIcons.erase(it);

}

}

delete ev;

return 0;

}

case WM_DEVICECHANGE:

{

auto* hdr = reinterpret_cast<DEV_BROADCAST_HDR*>(l);

if (w == DBT_DEVICEARRIVAL)

{

if (hdr && hdr->dbch_devicetype == DBT_DEVTYP_VOLUME)

{

Wh_Log(L"[MsgWnd WM_DEVICECHANGE] DBT_DEVTYP_VOLUME arrival");

auto* vol = reinterpret_cast<DEV_BROADCAST_VOLUME*>(l);

TrySpawnVolumeDialogForMask(vol->dbcv_unitmask);

}

else

{

Wh_Log(L"[MsgWnd WM_DEVICECHANGE] DBT_DEVICEARRIVAL type=%u",

hdr ? hdr->dbch_devicetype : 0);

}

}

else if (w == DBT_DEVICEREMOVECOMPLETE)

{

if (hdr && hdr->dbch_devicetype == DBT_DEVTYP_VOLUME)

{

auto* vol = reinterpret_cast<DEV_BROADCAST_VOLUME*>(l);

DWORD mask = vol->dbcv_unitmask;

for (int i = 0; i < 26; i++)

{

if (!(mask & (1u << i))) continue;

WCHAR szDrive[4] = { (WCHAR)(L'A' + i), L':', L'\\', L'\0' };

Wh_Log(L"[MsgWnd WM_DEVICECHANGE] Volume removed: %s", szDrive);

CloseAutoPlayDialogForDrive(L'A' + i); // close open dialog

UnregisterKnownDevice(szDrive); // allow re-trigger on next insert

CloseDialogKey(std::wstring(szDrive));

}

}

}

return TRUE;

}



}

return DefWindowProcW(hwnd, msg, w, l);

}



// ===========================================================================

// CM_Register_Notification callback

// (fires on a CM threadpool thread, posts to our message window)

// ===========================================================================

static DWORD CALLBACK CMCallback(

HCMNOTIFICATION /*hNotify*/,

PVOID pContext,

CM_NOTIFY_ACTION action,

PCM_NOTIFY_EVENT_DATA pData,

DWORD /*cbData*/)

{

HWND hwnd = reinterpret_cast<HWND>(pContext);

if (!hwnd) return ERROR_SUCCESS;

if (pData->FilterType != CM_NOTIFY_FILTER_TYPE_DEVICEINTERFACE)

return ERROR_SUCCESS;

if (action != CM_NOTIFY_ACTION_DEVICEINTERFACEARRIVAL &&

action != CM_NOTIFY_ACTION_DEVICEINTERFACEREMOVAL)

return ERROR_SUCCESS;



const GUID& cls = pData->u.DeviceInterface.ClassGuid;

bool isVol = IsEqualGUID(cls, AP_GUID_VOLUME);

bool isWPD = IsEqualGUID(cls, AP_GUID_WPD) || IsEqualGUID(cls, AP_GUID_USB);

if (!isVol && !isWPD) return ERROR_SUCCESS;



auto* ev = new (std::nothrow) DevArrival{

pData->u.DeviceInterface.SymbolicLink, isVol, isWPD

};

if (!ev) return ERROR_SUCCESS;



UINT uMsg = (action == CM_NOTIFY_ACTION_DEVICEINTERFACEARRIVAL)

? WM_AP_ARRIVED : WM_AP_REMOVED;



if (!PostMessageW(hwnd, uMsg, 0, reinterpret_cast<LPARAM>(ev)))

delete ev;



return ERROR_SUCCESS;

}



// ===========================================================================

// Windhawk entry points

// ===========================================================================

// ===========================================================================

// ShellExecuteExW hook intercepts right-click "Open AutoPlay..." verb

// ===========================================================================

using ShellExecuteExW_t = decltype(&ShellExecuteExW);

static ShellExecuteExW_t orig_ShellExecuteExW = nullptr;



static BOOL WINAPI hook_ShellExecuteExW(SHELLEXECUTEINFOW* psei)

{

if (psei && psei->lpVerb &&

_wcsicmp(psei->lpVerb, L"autoplay") == 0 &&

psei->lpFile && psei->lpFile[0])

{

WCHAR szDrive[MAX_PATH] = {};

StringCchCopyW(szDrive, ARRAYSIZE(szDrive), psei->lpFile);

size_t len = wcslen(szDrive);

if (len > 0 && szDrive[len-1] != L'\\') {

szDrive[len] = L'\\'; szDrive[len+1] = L'\0';

}

Wh_Log(L"[ShellExecuteExW] autoplay verb for %s -> Win7 dialog", szDrive);

UnregisterKnownDevice(szDrive); // force show even if already seen

SpawnVolumeDialog(szDrive, true);

return TRUE;

}

return orig_ShellExecuteExW ? orig_ShellExecuteExW(psei) : FALSE;

}



BOOL Wh_ModInit()

{

Wh_Log(L"Win7 AutoPlay Dialog mod initializing");

InitializeCriticalSection(&g_csKnownDevices);





g_hInstance = GetModuleHandleW(nullptr); // Explorer's hInstance



INITCOMMONCONTROLSEX icc = { sizeof(icc) };

icc.dwICC = ICC_LISTVIEW_CLASSES | ICC_LINK_CLASS |

ICC_PROGRESS_CLASS | ICC_STANDARD_CLASSES | ICC_WIN95_CLASSES;

InitCommonControlsEx(&icc);



// Register our message window class

WNDCLASSEXW wc = { sizeof(wc) };

wc.lpfnWndProc = MsgWndProc;

wc.hInstance = g_hInstance;

wc.lpszClassName = L"Win7AutoPlayMsgWnd";

RegisterClassExW(&wc);



// Use a normal (non-message-only) hidden window so it receives

// WM_DEVICECHANGE broadcasts from the OS. HWND_MESSAGE windows are

// excluded from broadcast messages (see deviceinstallux pattern).

g_hwndMsg = CreateWindowExW(0, L"Win7AutoPlayMsgWnd", nullptr,

WS_OVERLAPPED, 0, 0, 0, 0,

nullptr, nullptr, g_hInstance, nullptr);



if (!g_hwndMsg)

{

Wh_Log(L"Failed to create message window: %lu", GetLastError());

return FALSE;

}



// Belt-and-suspenders: also register for WM_DEVICECHANGE broadcasts

// (CM_Register_Notification alone misses some insertion events on Win10/11)

{

DEV_BROADCAST_DEVICEINTERFACE dbdi = {};

dbdi.dbcc_size = sizeof(dbdi);

dbdi.dbcc_devicetype = DBT_DEVTYP_DEVICEINTERFACE;

HDEVNOTIFY hdn = RegisterDeviceNotificationW(

g_hwndMsg, &dbdi,

DEVICE_NOTIFY_WINDOW_HANDLE | DEVICE_NOTIFY_ALL_INTERFACE_CLASSES);

if (hdn)

Wh_Log(L"RegisterDeviceNotification registered on message window");

else

Wh_Log(L"RegisterDeviceNotification failed: %lu", GetLastError());

}



// Initialize critical sections

EnsureDialogCS();

EnsureDialogsCS();



// Register IQueryCancelAutoPlay to suppress Explorer's own AutoPlay toast

RegisterCancelAutoPlay();



// Register CM notification for device arrivals

CM_NOTIFY_FILTER cmf = { sizeof(cmf) };

cmf.FilterType = CM_NOTIFY_FILTER_TYPE_DEVICEINTERFACE;

cmf.u.DeviceInterface.ClassGuid = GUID_NULL; // all interface classes



CONFIGRET cr = CM_Register_Notification(&cmf, g_hwndMsg, CMCallback, &g_hCMNotify);

if (cr != CR_SUCCESS)

Wh_Log(L"CM_Register_Notification failed: %u", cr);



// Hook DispatchMessageW as a third layer of device detection

// (catches WM_DEVICECHANGE already flowing through explorer's message loop)

Wh_SetFunctionHook((void*)DispatchMessageW,

(void*)hook_DispatchMessageW, (void**)&orig_DispatchMessageW);



// Hook ShellExecuteExW to intercept the right-click "Open AutoPlay..." verb

Wh_SetFunctionHook((void*)ShellExecuteExW,

(void*)hook_ShellExecuteExW, (void**)&orig_ShellExecuteExW);

Wh_Log(L"Hooked ShellExecuteExW for autoplay verb");



Wh_Log(L"Win7 AutoPlay Dialog mod loaded successfully");

return TRUE;

}



void Wh_ModUninit()

{

Wh_Log(L"Win7 AutoPlay Dialog mod unloading");

g_knownDevices.clear();

DeleteCriticalSection(&g_csKnownDevices);



// Unregister IQueryCancelAutoPlay

UnregisterCancelAutoPlay();



// Unregister CM notification

if (g_hCMNotify)

{

CM_Unregister_Notification(g_hCMNotify);

g_hCMNotify = nullptr;

}



// Destroy message window

if (g_hwndMsg)

{

DestroyWindow(g_hwndMsg);

g_hwndMsg = nullptr;

}



// Clean up device icon cache

for (auto& kv : g_deviceIcons)

if (kv.second) DestroyIcon(kv.second);

g_deviceIcons.clear();



// Clean up critical sections

if (g_csDialogKeysInit)

{

DeleteCriticalSection(&g_csDialogKeys);

g_csDialogKeysInit = false;

}

if (g_csDialogsInit)

{

DeleteCriticalSection(&g_csDialogs);

g_csDialogsInit = false;

}



Wh_Log(L"Win7 AutoPlay Dialog mod unloaded");

}
